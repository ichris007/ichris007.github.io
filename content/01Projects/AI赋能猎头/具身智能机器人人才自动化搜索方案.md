### 3.2 详细执行步骤

```python
# 伪代码：自动化搜索流程
def automated_robotics_talent_search(targets):

    # 步骤1: GitHub搜索
    github_results = []
    for target in targets:
        query = build_github_query(target)
        users = github_api.search_users(query)
        for user in users:
            profile = github_api.get_user_details(user)
            github_results.append(extract_profile_info(profile))

    # 步骤2: arXiv论文搜索
    arxiv_results = []
    for query in ARXIV_QUERIES:
        papers = arxiv_api.search(query)
        for paper in papers:
            authors = extract_authors(paper)
            arxiv_results.extend(authors)

    # 步骤3: 实验室官网搜索
    lab_results = []
    for lab_url in LAB_URLS:
        members_page = fetch_url(lab_url + "/People")
        members = parse_lab_members(members_page)
        lab_results.extend(members)

    # 步骤4: 数据整合
    all_results = merge_results(github_results, arxiv_results, lab_results)

    # 步骤5: 清洗和去重
    cleaned_results = clean_and_deduplicate(all_results)

    # 步骤6: 评分
    scored_results = score_candidates(cleaned_results)

    # 步骤7: 排序
    final_results = sort_by_score(scored_results)

    # 步骤8: 导出
    export_results(final_results, format="markdown")
    export_results(final_results, format="csv")

    return final_results
```

class AISearcher:
    """集成Claude AI进行智能搜索"""

    def __init__(self, api_key: str):
        self.api_key = api_key
        self.client = anthropic.Anthropic(api_key=api_key)

    def search_with_ai(self, prompt: str) -> Dict:
        """使用AI执行搜索任务"""

        message = self.client.messages.create(
            model="claude-3-5-sonnet-20241022",
            max_tokens=4000,
            messages=[
                {"role": "user", "content": prompt}
            ]
        )

        return self.parse_ai_response(message.content)

    def batch_search(self, prompts: List[str]) -> List[Dict]:
        """批量搜索"""
        results = []
        for prompt in prompts:
            result = self.search_with_ai(prompt)
            results.append(result)
        return results
